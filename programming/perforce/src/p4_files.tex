\section{p4 --- Working with Files}

\textit{Note:} \texttt{...} is a recursive wildcard matching all
files under a path (eg.\ \texttt{p4 revert ...} = everything under
cwd).

\subsubsection*{Sync}

\texttt{p4 sync [path...]}\\
Updates the workspace to match the depot, per the client view
(like \texttt{git pull} for files). With no path, acts on the
subtree under cwd.

\begin{tabularx}{\linewidth}{lX}
    \texttt{path\#<rev>}     & Sync to a specific revision.\\
    \texttt{path@<cl/label>} & Sync to the state as of a
                                 changelist, label, or date.\\
\hline
    \texttt{-f} & Force: re-transfer files even if already at that
                    revision (eg.\ to repair local edits made
                    outside p4).\\
    \texttt{-n} & Preview only --- no files transferred.\\
\hline
\end{tabularx}


\subsubsection*{Open for Change}

\begin{tabularx}{\linewidth}{lX}
    \texttt{p4 edit <file>} & Open an existing depot file for edit
                                (checkout).\\
    \texttt{p4 add <file>}  & Mark a new (untracked) file for add.\\
\hline
    \texttt{-c <cl>}        & Open in pending changelist
                                \texttt{<cl>} instead of
                                \texttt{default}. Applies to
                                \texttt{edit}, \texttt{add},
                                \texttt{revert}, \texttt{reopen},
                                \texttt{shelve}, \texttt{unshelve},
                                \texttt{submit}.\\
\hline
\end{tabularx}


\subsubsection*{Reconcile}

\texttt{p4 reconcile [-a] [-e] [-d] [-c <cl>] [path...]}\\
Detects changes made outside \texttt{p4} and opens files
accordingly. No flags = all of \texttt{-a -e -d}.

\begin{tabularx}{\linewidth}{lX}
    \texttt{-a} & Open files added on disk \texttt{for add}.\\
    \texttt{-e} & Open modified files \texttt{for edit}.\\
    \texttt{-d} & Open missing files \texttt{for delete}.\\
\hline
\end{tabularx}


\subsubsection*{Revert / Reopen}

\begin{tabularx}{\linewidth}{lX}
    \texttt{p4 revert [-a] [-c <cl>] <file>}
                & Discard pending changes, restore depot version.\\
    \texttt{-a} & Only revert files \textit{unchanged} from depot
                    (safe revert).\\
\hline
    \texttt{p4 reopen -c <cl> <file>}
                & Move opened file(s) to a different pending
                    changelist.\\
\hline
\end{tabularx}


\subsubsection*{Shelve / Unshelve}

\begin{tabularx}{\linewidth}{lX}
    \texttt{p4 shelve -c <cl> [file]}
                & Save files opened in \texttt{<cl>} to the server
                    without submitting.\\
    \texttt{-d} & With \texttt{shelve}: discard the shelved
                    file(s) from \texttt{<cl>}.\\
\hline
    \texttt{p4 unshelve [file]}
                & Restore shelved files into a pending
                    changelist.\\
    \texttt{-s <cl>}  & Source: changelist to unshelve from.\\
    \texttt{-c <cl2>} & Target: changelist to receive the files
                          (default \texttt{default}).\\
\hline
\end{tabularx}


\subsubsection*{Submit}

\begin{tabularx}{\linewidth}{lX}
    \texttt{p4 submit -c <cl>} & Submit pending changelist
                                    \texttt{<cl>} to the depot.\\
\hline
\end{tabularx}


\subsubsection*{Diff}

\texttt{p4 diff [file...]}\\
Compare the file \textbf{on disk} against a depot revision. Runs
\texttt{P4DIFF} (eg.\ \texttt{vimdiff}) if set. Default: opened
files vs the synced (\texttt{have}) revision.

\begin{tabularx}{\linewidth}{lX}
    \texttt{p4 diff file}      & Disk vs have revision.\\
    \texttt{p4 diff file\#2}   & Disk vs revision \texttt{\#2}.\\
    \texttt{p4 diff file@<cl>} & Disk vs the file as of changelist
                                   \texttt{<cl>}.\\
    \texttt{-f} & Diff files not opened for edit.\\
\hline
\end{tabularx}

\texttt{p4 diff2 <f1> <f2>}\\
Diff two \textbf{depot} revisions. Runs \textbf{on the server}, so
\texttt{P4DIFF} is ignored --- output is a plain text diff.

\begin{tabularx}{\linewidth}{lX}
    \texttt{p4 diff2 f\#1 f\#4} & Between two revisions of a
                                    file.\\
    \texttt{p4 diff2 f@<cl1> f@<cl2>}
                                & Same file across two
                                    changelists.\\
    \texttt{p4 diff2 f1 f2}     & Two different depot files.\\
    \texttt{-u} & Unified diff format.\\
    \texttt{-q} & Quiet: omit identical-file messages.\\
\hline
\end{tabularx}

To use \texttt{vimdiff} on two depot revisions, print them to
temp files:

\begin{mdframed}[backgroundcolor=gray!10,linecolor=Firebrick4]
\begin{verbatim}
$ p4 print -q file#1 > /tmp/a
$ p4 print -q file#4 > /tmp/b
$ vimdiff /tmp/a /tmp/b
\end{verbatim}
\end{mdframed}


\subsubsection*{Revision Specifiers}

\begin{tabularx}{\linewidth}{lX}
    \texttt{file\#<n>}  & Revision \texttt{n}.\\
    \texttt{file\#head} & Latest depot revision.\\
    \texttt{file\#have} & Revision last synced into the
                            workspace.\\
    \texttt{file@<cl>}  & Revision as of changelist \texttt{<cl>}
                            (also takes a label or date).\\
    \texttt{file@=<cl>} & Revision shelved or submitted in exactly
                            \texttt{<cl>}.\\
\hline
\end{tabularx}


\subsubsection*{Inspecting}

\begin{tabularx}{\linewidth}{lX}
    \texttt{p4 opened}
                & List opened (checked-out) files.\\
    \texttt{-c <cl>}   & Filter by changelist.\\
    \texttt{-u <user>} & Filter by user.\\
\hline
    \texttt{p4 filelog <file>}
                & Show a file's revision history.\\
\hline
    \texttt{p4 changes}
                & List changelists.\\
    \texttt{-u <user>}   & Filter by user.\\
    \texttt{-m <n>}      & Limit output to \texttt{n} changelists.\\
    \texttt{-s <status>} & Filter by status: \texttt{pending} /
                             \texttt{submitted} /
                             \texttt{shelved}.\\
\hline
\end{tabularx}
