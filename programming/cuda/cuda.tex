\section*{Function Qualifiers}

\begin{tabularx}{\linewidth}{lX}
\texttt{\_\_global\_\_}               & Kernel func. Returns void. \\
\texttt{\_\_device\_\_}               & Function callable only from device. \\
\texttt{\_\_host\_\_}                 & Function callable from host (default). \\
\texttt{\_\_host\_\_ \_\_device\_\_}  & Compiled for both host and device.\\
\end{tabularx}

\textbf{Kernel Launch:} \texttt{}
\begin{mdframed}[nobreak=true,backgroundcolor=gray!10,linecolor=Firebrick4]
\begin{verbatim}
kernel<<<gridDim, blockDim, sharedMem, stream>>>(args);
\end{verbatim}
\end{mdframed}


\begin{tabularx}{\linewidth}{lX}
\texttt{gridDim}        & Grid size (int or dim3). \\
\texttt{blockDim}       & Block size (int or dim3). \\
\texttt{sharedMem}      & Dynamic shared memory bytes (optional). \\
\texttt{stream}         & CUDA stream (optional). \\
\end{tabularx}

\begin{mdframed}[nobreak=true,backgroundcolor=magenta!10,linecolor=magenta]
\texttt{\_\_device\_\_} To declare global variable in the device.
\end{mdframed}

\section*{Built-in Variables}

\begin{tabularx}{\linewidth}{lX}
\texttt{gridDim}      & Dimensions of the grid (\texttt{dim3}). \\
\texttt{gridDim.x/y/z}  & Number of blocks in each dimension. \\
\texttt{blockDim}     & Dimensions of each block (\texttt{dim3}). \\
\texttt{blockDim.x/y/z} & Number of threads in each dimension. \\
\texttt{blockIdx}     & Block index within the grid (\texttt{uint3}). \\
\texttt{blockIdx.x/y/z} & Current block's index in each dimension. \\
\texttt{threadIdx}    & Thread index within the block (\texttt{uint3}). \\
\texttt{threadIdx.x/y/z} & Current thread's index in each dimension. \\
\texttt{warpSize}     & Threads per warp (constant = 32). \\
\end{tabularx}

\textbf{Global Thread ID:}
\begin{mdframed}[nobreak=true,backgroundcolor=gray!10,linecolor=Firebrick4]
\begin{verbatim}
int tid = blockIdx.x * blockDim.x + threadIdx.x; // 1D
int col = blockIdx.x * blockDim.x + threadIdx.x; // 2D
int row = blockIdx.y * blockDim.y + threadIdx.y;
\end{verbatim}
\end{mdframed}


\section*{Memory Management}

% Memory allocation.
\begin{tabularx}{\linewidth}{lX}
\hline
\texttt{cudaMalloc(\&ptr, size)} & Allocate device memory. \\
                                 & \textcolor{red}{NOTE: ptr to ptr.}\\
\texttt{cudaFree(ptr)} & Free device memory. \\
\hline
\end{tabularx}

\begin{tabularx}{\linewidth}{lX}
\texttt{cudaHostAlloc(\&ptr, size, flags)} & Pinned host memory (faster transfers) \\
\texttt{cudaFreeHost(ptr)} & Free pinned host memory \\
\texttt{cudaMallocManaged(\&ptr, size)} & Unified memory (host \& device accessible) \\
\hline
\end{tabularx}

% Memory transfer.
\begin{tabularx}{\linewidth}{lX}
\texttt{cudaMemcpy(dst, src, size, kind)} &  \\
\end{tabularx}
\begin{tabularx}{\linewidth}{lX}
\texttt{dst}                              & Pointers.\\
\texttt{src}                              & \\
\textit{kind}                             &  \texttt{cudaMemcpyHostToDevice}.\\
                                          &  \texttt{cudaMemcpyDeviceToHost}.\\
                                          &  \texttt{cudaMemcpyDeviceToDevice}.\\
\texttt{size}                             &  Eg: \texttt{n * sizeof(int)}.\\
\hline
\end{tabularx}

\begin{tabularx}{\linewidth}{lX}
\texttt{cudaMemcpyAsync(..., stream)} & Async copy. Requires pinned memory \\
\hline
\end{tabularx}


\subsubsection*{Memory Types}

\begin{tabularx}{\linewidth}{llX}
\hline
\textbf{Type} & \textbf{Scope} & \textbf{Description} \\
\hline
Registers     & Thread   & Fastest; automatic allocation. \\
Local         & Thread   & Spills from registers; stored in global memory. \\
Shared        & Block    & Fast; \texttt{\_\_shared\_\_} qualifier. \\
Global        & Grid     & Slowest; \texttt{cudaMalloc} or \texttt{\_\_device\_\_}. \\
Constant      & Grid     & Read-only; cached; \texttt{\_\_constant\_\_} (64KB max). \\
Texture       & Grid     & Read-only; cached; optimized for 2D spatial locality. \\
\hline
\end{tabularx}


\section*{Synchronization}

\subsection*{Synchronization Primitives}

\begin{tabularx}{\linewidth}{lX}
\hline
\textbf{Function} & \textbf{Description} \\
\hline
\texttt{\_\_syncthreads()}       & Block-level barrier; all threads in block must reach. \\
\texttt{\_\_syncwarp(mask)}      & Warp-level sync; mask specifies participating threads. \\
\texttt{cudaDeviceSynchronize()} & Host waits for all device work to complete. \\
\texttt{cudaStreamSynchronize(s)} & Host waits for all work in stream \texttt{s}. \\
\texttt{cudaEventSynchronize(e)} & Host waits for event \texttt{e} to complete. \\
\hline
\end{tabularx}

\subsubsection*{Memory Fences}

\begin{tabularx}{\linewidth}{lX}
\texttt{\_\_threadfence\_block()} & Ensures writes visible to block threads. \\
\texttt{\_\_threadfence()}        & Ensures writes visible to all device threads. \\
\texttt{\_\_threadfence\_system()} & Ensures writes visible to host and device. \\
\end{tabularx}

\subsection*{Atomic Operations}

\begin{tabularx}{\linewidth}{lX}
\hline
\textbf{Function} & \textbf{Operation} \\
\hline
\texttt{atomicAdd(addr, val)}  & \texttt{*addr += val}; returns old value. \\
\texttt{atomicSub(addr, val)}  & \texttt{*addr -= val}; returns old value. \\
\texttt{atomicExch(addr, val)} & \texttt{*addr = val}; returns old value. \\
\texttt{atomicMin(addr, val)}  & \texttt{*addr = min(*addr, val)}. \\
\texttt{atomicMax(addr, val)}  & \texttt{*addr = max(*addr, val)}. \\
\texttt{atomicInc(addr, val)}  & \texttt{*addr = (*addr >= val) ? 0 : *addr+1}. \\
\texttt{atomicDec(addr, val)}  & \texttt{*addr = (*addr==0 || *addr>val) ? val : *addr-1}. \\
\texttt{atomicCAS(addr, cmp, val)} & If \texttt{*addr == cmp}, set \texttt{*addr = val}; returns old. \\
\hline
\end{tabularx}

\subsubsection*{Bitwise Atomics}

\begin{tabularx}{\linewidth}{lX}
\texttt{atomicAnd(addr, val)} & \texttt{*addr \&= val}. \\
\texttt{atomicOr(addr, val)}  & \texttt{*addr |= val}. \\
\texttt{atomicXor(addr, val)} & \texttt{*addr \^{}= val}. \\
\end{tabularx}

\subsection*{Events}

\begin{tabularx}{\linewidth}{lX}
\hline
\texttt{cudaEvent\_t}            & Event handle type. \\
\texttt{cudaEventCreate(\&e)}    & Create an event. \\
\texttt{cudaEventDestroy(e)}     & Destroy an event. \\
\texttt{cudaEventRecord(e, s)}   & Record event in stream \texttt{s}. \\

\textbf{Timing with events:}
\begin{mdframed}[nobreak=true,backgroundcolor=gray!10,linecolor=Firebrick4]
\begin{verbatim}
cudaEventRecord(start, stream);
kernel<<<...>>>();
cudaEventRecord(stop, stream);
cudaEventSynchronize(stop);
float ms;
cudaEventElapsedTime(&ms, start, stop);
\end{verbatim}
\end{mdframed}

\subsection*{Stream Priorities}

\begin{tabularx}{\linewidth}{lX}
\texttt{cudaDeviceGetStreamPriorityRange(\&lo, \&hi)} & Get priority range. \\
\texttt{cudaStreamCreateWithPriority(\&s, flags, pri)} & Create with priority. \\
\end{tabularx}

\begin{itemize}[leftmargin=*,nosep]
\item Lower number = higher priority.
\item Work from higher-priority streams may preempt lower.
\end{itemize}

\section*{Support}

\subsection*{Error Handling}

\begin{tabularx}{\linewidth}{lX}
\hline
\texttt{cudaError\_t}        & Error type returned by CUDA calls. \\
\texttt{cudaSuccess}         & No error (value 0). \\
\texttt{cudaGetLastError()}  & Returns and resets last error. \\
\texttt{cudaPeekAtLastError()} & Returns last error without reset. \\
\texttt{cudaGetErrorString(err)} & Error code to string. \\
\texttt{cudaGetErrorName(err)}   & Error code to name. \\
\hline
\end{tabularx}

\textbf{Error checking pattern:}
\begin{mdframed}[nobreak=true,backgroundcolor=gray!10,linecolor=Firebrick4]
\begin{verbatim}
cudaError_t err = cudaMalloc(&ptr, size);
if (err != cudaSuccess) {
    printf("Error: %s\n", cudaGetErrorString(err));
}
// After kernel launch (async, doesn't return error):
kernel<<<grid, block>>>();
err = cudaGetLastError();       // Check launch error
err = cudaDeviceSynchronize();  // Check execution error
\end{verbatim}
\end{mdframed}

\subsection*{NVCC Compiler Options}

\begin{tabularx}{\linewidth}{lX}
\hline
\textbf{Option} & \textbf{Description} \\
\hline
\texttt{-arch=sm\_XX}        & Target GPU architecture (e.g., \texttt{sm\_86}). \\
\texttt{-gencode=arch=compute\_XX,code=sm\_XX} & Generate code for specific arch. \\
\texttt{-G}                  & Debug mode; disables optimizations. \\
\texttt{-lineinfo}           & Include line info for profiling. \\
\texttt{-O0/-O1/-O2/-O3}     & Optimization level. \\
\texttt{-Xptxas -v}          & Print register/memory usage per kernel. \\
\texttt{-maxrregcount=N}     & Limit registers per thread. \\
\texttt{--ptxas-options=-v}  & Verbose PTX assembler output. \\
\texttt{-res-usage}          & Report resource usage. \\
\texttt{-Xcompiler "opts"}   & Pass options to host compiler. \\
\hline
\end{tabularx}

\textbf{Example compilation:}
\begin{mdframed}[nobreak=true,backgroundcolor=gray!10,linecolor=Firebrick4]

nvcc -arch=sm_86 -O3 -Xptxas -v -o prog prog.cu
\end{verbatim}
